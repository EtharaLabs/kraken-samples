+    def _unitary_(self) -> np.ndarray | NotImplementedType:
+        """Returns the unitary of a single-qubit circuit operation, if it has one.
+
+        The matrices of the contained operations are multiplied directly, which avoids
+        decomposing the operation into a mapped circuit. Operations on more than one
+        qubit, operations with symbolic repetitions or `repeat_until`, and circuits
+        containing a non-unitary operation return NotImplemented, so that the unitary
+        protocol falls back to its other strategies.
+        """
+        qubits = self.qubits
+        if len(qubits) != 1 or self.repeat_until or not isinstance(self.repetitions, INT_CLASSES):
+            return NotImplemented
+        mat = np.eye(qubits[0].dimension, dtype=np.complex128)
+        for op in self.circuit.all_operations():
+            if self.param_resolver:
+                op = protocols.resolve_parameters(op, self.param_resolver, recursive=False)
+            op_unitary = protocols.unitary(op, None)
+            if op_unitary is None:
+                return NotImplemented
+            # Operations without qubits contribute a global phase.
+            mat = op_unitary @ mat if op.qubits else op_unitary[0, 0] * mat
+        if self.repetitions < 0:
+            mat = mat.conj().T
+        return np.linalg.matrix_power(mat, abs(int(self.repetitions)))
