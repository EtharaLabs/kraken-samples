+def _string_missing_mask(ar):
+    """
+    Mask of the missing values in a string array that do not compare like
+    strings, or None if the array cannot hold such values.
+    """
+    if not hasattr(ar.dtype, "na_object"):
+        return None
+    if isinstance(ar.dtype.na_object, str):
+        # String sentinels compare (and sort) like the string itself.
+        return None
+    na = np.array(ar.dtype.na_object, dtype=ar.dtype)
+    if np.isnan(na):
+        return np.isnan(ar)
+    # Missing values that are not NaN-like compare equal to each other.
+    return ar == na

+def _isin_strings(ar1, ar2, invert=False):
+    """
+    `_isin` for unicode and StringDType arrays, giving the same result as
+    comparing `ar1` with each element of `ar2` in turn.
+    """
+    if invert:

+    else:
+        mask = np.zeros(len(ar1), dtype=bool)
+
+    missing2 = _string_missing_mask(ar2)
+    if missing2 is not None and missing2.any():
+        # All missing values in `ar2` are the same sentinel object, so a
+        # single comparison accounts for them.
+        if invert:
+            mask &= (ar1 != ar2.dtype.na_object)
+        else:
+            mask |= (ar1 == ar2.dtype.na_object)
+        ar2 = ar2[~missing2]
+

+    # values need to be compared.  These hold no missing values any more and
+    # can be cast to a common dtype which supports sorting.
