+def _sparse_norm2_cols(X, X_mean, sample_weight):
+    """Squared norms of the columns of the centered sparse X.
+
+    Parameters
+    ----------
+    X : sparse matrix of shape (n_samples, n_features) in CSC format
+        Training data.
+
+    X_mean : ndarray of shape (n_features,)
+        Center of the columns of X.
+
+    sample_weight : None or ndarray of shape (n_samples,)
+        Sample weights. If None, all samples have weight one.
+
+    Returns
+    -------
+    norm2_cols_X : ndarray of shape (n_features,)
+        `np.sum(sample_weight * (X - X_mean)**2, axis=0)`, computed without
+        densifying X.
+    """
+    n_samples, n_features = X.shape
+    nnz_cols = np.diff(X.indptr)
+    cols = np.repeat(np.arange(n_features), nnz_cols)  # column of each nonzero
+    X_mean_nz = X_mean[cols]
+    # The zero entries of column j contribute X_mean[j] ** 2 each.
+    if sample_weight is None:
+        norm2_cols_X = np.bincount(
+            cols, weights=(X.data - X_mean_nz) ** 2, minlength=n_features
+        ) + (n_samples - nnz_cols) * X_mean**2
+    else:
+        norm2_cols_X = np.bincount(
+            cols,
+            weights=sample_weight[X.indices]
+            * ((X.data - X_mean_nz) ** 2 - X_mean_nz**2),
+            minlength=n_features,
+        ) + np.sum(sample_weight) * X_mean**2
+    return norm2_cols_X.astype(X.dtype, copy=False)
